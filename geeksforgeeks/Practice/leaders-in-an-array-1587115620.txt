class Solution {
  public:
    vector<int> leaders(vector<int>& arr) {
        vector<int> result;
        int n = arr.size();
        
        if (n == 0) {
            return result;
        }
        
        int max_right = arr[n - 1];
        result.push_back(max_right);
        
        for (int i = n - 2; i >= 0; i--) {
            if (arr[i] >= max_right) {
                max_right = arr[i];
                result.push_back(max_right);
            }
        }
        
        reverse(result.begin(), result.end());
        
        return result;
    }
};
